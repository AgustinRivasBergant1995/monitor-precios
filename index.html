montar = function(cont, getOption, controles) {
  requestAnimationFrame(() => {
    const chart = window.echarts.init(cont);
    const dibujar = () => chart.setOption(getOption(), true);
    (controles || []).forEach(c => c.addEventListener("input", dibujar));
    dibujar();
    new ResizeObserver(() => chart.resize()).observe(cont);
  });
}

ddmm = (f) => { const p = f.split("-"); return p[2] + "/" + p[1]; }

baseLineas = function(esAcum, colores, labels, seriesArr) {
  return {
    color: colores,
    textStyle: {fontFamily: FUENTE_G, color: "#1e2b2f"},
    grid: {left: 66, right: 22, top: 38, bottom: 34},
    legend: {top: 4, icon: "roundRect", itemWidth: 14, itemHeight: 8, textStyle: {color: "#45606b"}, type: "scroll"},
    tooltip: {trigger: "axis", valueFormatter: (v) => v == null ? "–" : (esAcum ? (+v).toFixed(2) : (v >= 0 ? "+" : "") + (+v).toFixed(2) + "%")},
    xAxis: {type: "category", data: labels, boundaryGap: false, axisLine: {lineStyle: {color: "#d4cfc6"}}, axisTick: {show: false}, axisLabel: {color: "#5b727c"}},
    yAxis: {type: "value", scale: true, name: esAcum ? "Índice (base 100)" : "Variación semanal (%)",
            nameLocation: "middle", nameGap: 48, nameRotate: 90, nameTextStyle: {color: "#8a9aa2", fontSize: 12},
            axisLabel: {color: "#5b727c"}, splitLine: {lineStyle: {color: "#ece7de"}}},
    series: seriesArr.map(s => ({
      name: s.name, type: "line", smooth: true, symbol: "circle", symbolSize: 7, lineStyle: {width: 3}, emphasis: {focus: "series"},
      markLine: esAcum ? undefined : {silent: true, symbol: "none", lineStyle: {color: "#c9d6dd", type: "dashed"}, data: [{yAxis: 0}]},
      data: s.data
    }))
  };
}

graficoLineas = function(datos, cats, colores, capN, capT) {
  const fechasISO = [...new Set(datos.map(d => d.fecha))].sort();
  const labels = fechasISO.map(ddmm);
  const val = (cat, campo, f) => { const r = datos.find(d => d.cat === cat && d.fecha === f); return r && r[campo] != null ? +r[campo] : null; };
  const tog = Inputs.radio(["Acumulado", "Variación semanal"], {value: "Acumulado"});
  tog.classList.add("pill-toggle");
  const cont = html`<div style="width:100%;height:370px;"></div>`;
  const cap = html`<div class="grafico-caption"><b>Gráfico ${capN}.</b> ${capT}</div>`;
  const getOption = () => {
    const esAcum = tog.value === "Acumulado";
    const campo = esAcum ? "acumulado" : "variacion";
    return baseLineas(esAcum, cats.map(c => colores[c]), labels, cats.map(c => ({name: c, data: fechasISO.map(f => val(c, campo, f))})));
  };
  montar(cont, getOption, [tog]);
  return html`<div>${tog}${cont}${cap}</div>`;
}

graficoDesplegable = function(datos, colores, capN, capT) {
  const fechasISO = [...new Set(datos.map(d => d.fecha))].sort();
  const labels = fechasISO.map(ddmm);
  const cats = ["Total (ponderado)"].concat([...new Set(datos.map(d => d.cat))].filter(c => c !== "Total (ponderado)").sort());
  const sel = Inputs.select(cats, {label: "Subcategoría", value: "Total (ponderado)"});
  const tog = Inputs.radio(["Acumulado", "Variación semanal"], {value: "Acumulado"});
  tog.classList.add("pill-toggle");
  const cont = html`<div style="width:100%;height:360px;"></div>`;
  const cap = html`<div class="grafico-caption"><b>Gráfico ${capN}.</b> ${capT}</div>`;
  const getOption = () => {
    const esAcum = tog.value === "Acumulado";
    const campo = esAcum ? "acumulado" : "variacion";
    const cat = sel.value;
    const color = colores[cat] || "#1f5673";
    const dd = fechasISO.map(f => { const r = datos.find(d => d.cat === cat && d.fecha === f); return r && r[campo] != null ? +r[campo] : null; });
    return {
      color: [color], textStyle: {fontFamily: FUENTE_G, color: "#1e2b2f"},
      grid: {left: 66, right: 22, top: 20, bottom: 34},
      tooltip: {trigger: "axis", valueFormatter: (v) => v == null ? "–" : (esAcum ? (+v).toFixed(2) : (v >= 0 ? "+" : "") + (+v).toFixed(2) + "%")},
      xAxis: {type: "category", data: labels, boundaryGap: false, axisLine: {lineStyle: {color: "#d4cfc6"}}, axisTick: {show: false}, axisLabel: {color: "#5b727c"}},
      yAxis: {type: "value", scale: true, name: esAcum ? "Índice (base 100)" : "Variación semanal (%)",
              nameLocation: "middle", nameGap: 48, nameRotate: 90, nameTextStyle: {color: "#8a9aa2", fontSize: 12},
              axisLabel: {color: "#5b727c"}, splitLine: {lineStyle: {color: "#ece7de"}}},
      series: [{type: "line", smooth: true, symbol: "circle", symbolSize: 8, lineStyle: {width: 3}, areaStyle: {opacity: 0.12},
                markLine: esAcum ? undefined : {silent: true, symbol: "none", lineStyle: {color: "#c9d6dd", type: "dashed"}, data: [{yAxis: 0}]},
                data: dd}]
    };
  };
  montar(cont, getOption, [sel, tog]);
  return html`<div style="display:flex;gap:16px;flex-wrap:wrap;align-items:center;margin-bottom:12px;">${tog}${sel}</div>${cont}${cap}`;
}

graficoBarras = function(datos, capN, capT, opts) {
  const {xlabel = "Variación semanal (%)", suf = "%"} = (opts || {});
  const orden = datos.slice().sort((a, b) => a.valor - b.valor);
  const cats = orden.map(d => d.cat);
  const vals = orden.map(d => +d.valor);
  const cont = html`<div style="width:100%;height:${Math.max(200, datos.length * 34 + 70)}px;"></div>`;
  const cap = html`<div class="grafico-caption"><b>Gráfico ${capN}.</b> ${capT}</div>`;
  const getOption = () => ({
    textStyle: {fontFamily: FUENTE_G, color: "#1e2b2f"},
    grid: {left: 200, right: 40, top: 12, bottom: 44},
    tooltip: {trigger: "axis", axisPointer: {type: "shadow"}, valueFormatter: (v) => (v >= 0 ? "+" : "") + (+v).toFixed(2) + suf},
    xAxis: {type: "value", name: xlabel, nameLocation: "middle", nameGap: 30, axisLabel: {color: "#5b727c", formatter: (v) => (v >= 0 ? "+" : "") + v + suf},
            splitLine: {lineStyle: {color: "#ece7de"}}, axisLine: {show: false}},
    yAxis: {type: "category", data: cats, axisLabel: {color: "#5b727c"}, axisLine: {lineStyle: {color: "#d4cfc6"}}, axisTick: {show: false}},
    series: [{type: "bar", barWidth: "62%",
              data: vals.map(v => ({value: v, itemStyle: {color: v >= 0 ? SUBA : BAJA, borderRadius: v >= 0 ? [0, 4, 4, 0] : [4, 0, 0, 4]}}))}]
  });
  montar(cont, getOption, []);
  return html`<div>${cont}${cap}</div>`;
}

graficoHeatmap = function(datos, capN, capT) {
  const xdom = [...new Set(datos.map(d => d.fecha))];
  const suma = {}; datos.forEach(d => suma[d.cat] = (suma[d.cat] || 0) + (+d.var));
  const ydom = Object.keys(suma).sort((a, b) => suma[a] - suma[b]);
  const M = Math.max(...datos.map(d => Math.abs(+d.var)), 0.1);
  const data = datos.map(d => [xdom.indexOf(d.fecha), ydom.indexOf(d.cat), +(+d.var).toFixed(2)]);
  const cont = html`<div style="width:100%;height:${ydom.length * 28 + 140}px;"></div>`;
  const cap = html`<div class="grafico-caption"><b>Gráfico ${capN}.</b> ${capT}</div>`;
  const getOption = () => ({
    textStyle: {fontFamily: FUENTE_G, color: "#1e2b2f"},
    grid: {left: 200, right: 26, top: 18, bottom: 74},
    tooltip: {position: "top", formatter: (p) => ydom[p.value[1]] + " · " + xdom[p.value[0]] + ": " + (p.value[2] >= 0 ? "+" : "") + p.value[2] + "%"},
    xAxis: {type: "category", data: xdom, axisLabel: {color: "#5b727c"}, axisLine: {lineStyle: {color: "#d4cfc6"}}, axisTick: {show: false}},
    yAxis: {type: "category", data: ydom, axisLabel: {color: "#5b727c"}, axisLine: {lineStyle: {color: "#d4cfc6"}}, axisTick: {show: false}},
    visualMap: {min: -M, max: M, calculable: true, orient: "horizontal", left: "center", bottom: 12,
                inRange: {color: [BAJA, "#f7f4ef", SUBA]}, text: ["sube", "baja"], textStyle: {color: "#5b727c"}},
    series: [{type: "heatmap", data: data, label: {show: true, formatter: (p) => p.value[2].toFixed(1), fontSize: 9, color: "#2a2a2a"},
              itemStyle: {borderColor: "#fff", borderWidth: 1}, emphasis: {itemStyle: {borderColor: "#1e2b2f", borderWidth: 1.5}}}]
  });
  montar(cont, getOption, []);
  return html`<div>${cont}${cap}</div>`;
}

graficoHistograma = function(datos, capN, capT) {
  const min = -30, max = 30, w = 2, nb = (max - min) / w;
  const bs = new Array(nb).fill(0), bb = new Array(nb).fill(0);
  datos.forEach(d => { const i = Math.min(nb - 1, Math.max(0, Math.floor((+d.delta - min) / w))); if (d.signo === "Subió") bs[i]++; else bb[i]++; });
  const centros = Array.from({length: nb}, (_, i) => (min + w * i + w / 2).toFixed(0));
  const cont = html`<div style="width:100%;height:320px;"></div>`;
  const cap = html`<div class="grafico-caption"><b>Gráfico ${capN}.</b> ${capT}</div>`;
  const getOption = () => ({
    textStyle: {fontFamily: FUENTE_G, color: "#1e2b2f"},
    grid: {left: 56, right: 20, top: 34, bottom: 44},
    legend: {top: 4, textStyle: {color: "#45606b"}},
    tooltip: {trigger: "axis", axisPointer: {type: "shadow"}},
    xAxis: {type: "category", data: centros, name: "Variación del precio (%)", nameLocation: "middle", nameGap: 28,
            axisLabel: {color: "#5b727c"}, axisLine: {lineStyle: {color: "#d4cfc6"}}, axisTick: {show: false}},
    yAxis: {type: "value", name: "Cantidad de SKU", nameLocation: "middle", nameGap: 40, nameRotate: 90, nameTextStyle: {color: "#8a9aa2"},
            axisLabel: {color: "#5b727c"}, splitLine: {lineStyle: {color: "#ece7de"}}},
    series: [
      {name: "Subió", type: "bar", stack: "x", data: bs, itemStyle: {color: SUBA}, barCategoryGap: "8%"},
      {name: "Bajó", type: "bar", stack: "x", data: bb, itemStyle: {color: BAJA}, barCategoryGap: "8%"}
    ]
  });
  montar(cont, getOption, []);
  return html`<div>${cont}${cap}</div>`;
}

Metodología

Cómo se construye el índice

1

Relevamiento

Scraping semanal de La Anónima y El Todo, clasificado en COICOP con apertura nivel Clase. Se excluyen del relevamiento bienes durables comercializados por los supermercados, como electrodomesticos, indumentaria, articulos de ferretería, esparcimiento, educación, entre otros.

2

Índice elemental

Jevons (media geométrica de relativos) por subcategoría, encadenado (base 100). Los precios faltantes son rellenados con la semana contigua.

3

Ponderación

El índice general y el de alimentos agregan los elementales como promedio ponderado por la canasta correspondiente. Las ponderaciones se construyen con los microdatos de la Encuesta Nacional de Gasto de los Hogares 2017/2018 (ENGHo 2017/2018).

4

Actualización

Todos los jueves.
